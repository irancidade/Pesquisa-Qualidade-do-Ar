package net.sourceforge.jFuzzyLogic.demo.tipper;

import net.sourceforge.jFuzzyLogic.FIS;

public class TesteQualidade {
	public static void main(String[] args) throws Exception {
	    FIS fis = FIS.load("fcl/QualidadeAr.fcl", true); // Load from 'FCL' file
	    fis.setVariable("pm10", 119.27); // Set inputs
	    fis.setVariable("pm25", 119.27);
	    fis.setVariable("co",  500);
	    fis.setVariable("co2",104);
	    fis.evaluate(); // Evaluate

	    // Show output variable
	    System.out.println("Output value:" + fis.getVariable("quality").getValue()); 
	  }
	}
